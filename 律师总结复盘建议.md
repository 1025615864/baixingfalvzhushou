# app/domains/firm/state_machine.py
from enum import Enum
from typing import Optional

class FirmStatus(str, Enum):
    DRAFT = "draft"                     # 草稿
    PENDING_VERIFICATION = "pending"    # 待审核
    UNDER_REVIEW = "under_review"       # 审核中
    APPROVED = "approved"               # 已通过
    REJECTED = "rejected"               # 已拒绝
    SUSPENDED = "suspended"             # 已暂停
    DEACTIVATED = "deactivated"         # 已停用

# 合法的状态转换定义
FIRM_TRANSITIONS: dict[FirmStatus, set[FirmStatus]] = {
    FirmStatus.DRAFT: {
        FirmStatus.PENDING_VERIFICATION,   # 提交审核
    },
    FirmStatus.PENDING_VERIFICATION: {
        FirmStatus.UNDER_REVIEW,           # 开始审核
        FirmStatus.DRAFT,                  # 退回修改
    },
    FirmStatus.UNDER_REVIEW: {
        FirmStatus.APPROVED,               # 审核通过
        FirmStatus.REJECTED,               # 审核拒绝
    },
    FirmStatus.REJECTED: {
        FirmStatus.PENDING_VERIFICATION,   # 重新提交
    },
    FirmStatus.APPROVED: {
        FirmStatus.SUSPENDED,              # 暂停营业
        FirmStatus.DEACTIVATED,            # 主动停用
    },
    FirmStatus.SUSPENDED: {
        FirmStatus.APPROVED,               # 恢复营业
        FirmStatus.DEACTIVATED,            # 永久停用
    },
    FirmStatus.DEACTIVATED: set(),         # 终态，不可变更
}

class FirmStateMachine:
    """律所状态机"""

    @staticmethod
    def can_transition(current: FirmStatus, target: FirmStatus) -> bool:
        allowed = FIRM_TRANSITIONS.get(current, set())
        return target in allowed

    @staticmethod
    def transition(
        firm: LawFirm,
        target: FirmStatus,
        operator_id: str,
        reason: Optional[str] = None,
    ) -> LawFirm:
        current = FirmStatus(firm.status)
        if not FirmStateMachine.can_transition(current, target):
            raise BusinessError(
                "L30501",
                f"律所状态不允许从 [{current.value}] 变更为 [{target.value}]"
            )
        firm.status = target.value
        firm.status_changed_at = datetime.utcnow()
        firm.status_changed_by = operator_id
        firm.status_reason = reason
        return firm

# ✅ 服务层使用
class FirmAdminService:
    async def approve_firm(self, firm_id: UUID, admin_id: str, db: AsyncSession):
        async with self._tx.transaction() as session:
            firm = await session.get(LawFirm, firm_id, with_for_update=True)
            if not firm:
                raise NotFoundError("L40401", "律所不存在")

            # 状态机自动校验合法性
            FirmStateMachine.transition(
                firm,
                FirmStatus.APPROVED,
                operator_id=admin_id,
                reason="资质审核通过"
            )

            # 写入审核记录
            session.add(LawFirmVerification(
                firm_id=firm_id,
                action="approved",
                operator_id=admin_id,
            ))

            # Outbox 事件
            session.add(OutboxEvent(
                event_type="firm.approved",
                payload={"firm_id": str(firm_id)},
            ))

# app/domains/firm/events.py
class FirmDomainEvents:
    """律所领域事件 → 触发其他领域的级联操作"""

    @staticmethod
    async def on_firm_suspended(firm_id: UUID, db: AsyncSession, outbox: OutboxService):
        """律所暂停时的级联处理"""

        # 1. 暂停该律所所有律师的排班
        await db.execute(
            update(LawyerSchedule)
            .where(LawyerSchedule.lawyer_id.in_(
                select(Lawyer.id).where(Lawyer.firm_id == firm_id)
            ))
            .where(LawyerSchedule.date >= date.today())
            .where(LawyerSchedule.is_booked == False)
            .values(status="cancelled")
        )

        # 2. 通知已预约的用户
        affected_appointments = await db.execute(
            select(LawyerConsultation)
            .join(Lawyer)
            .where(Lawyer.firm_id == firm_id)
            .where(LawyerConsultation.status == "confirmed")
            .where(LawyerConsultation.scheduled_at >= datetime.utcnow())
        )
        for apt in affected_appointments.scalars():
            await outbox.save_event(
                "appointment.affected_by_firm_suspension",
                {
                    "appointment_id": str(apt.id),
                    "user_id": str(apt.user_id),
                    "lawyer_id": str(apt.lawyer_id),
                    "reason": "律所暂停营业",
                }
            )

    ("/api/v1/legal/admin", "*"):               (0, 60),    # 非管理员禁止
}

class RolAwareRateLimiter:
    async def check(self, request: Request, user_role: str, redis: Redis) -> bool:
        path = request.url.path
        user_id = request.state.user_id

        for (prefix, role), (limit, window) in RATE_LIMITS.items():
            if path.startswith(prefix) and (role == user_role or role == "*"):
                if limit == 0:
                    return False  # 禁止访问
                key = f"rate:{prefix}:{user_id}"
                count = await redis.incr(key)
                if count == 1:
                    await redis.expire(key, window)
                return count <= limit

        # 默认限流
        return True
