class LegalServicer(legal_pb2_grpc.LegalServiceServicer):
    def __init__(self, session_factory, cache_service):
        self._session_factory = session_factory
        self._cache = cache_service

    async def GetLawyer(self, request, context):
        # 先查缓存
        cached = await self._cache.get_lawyer(request.lawyer_id)
        if cached:
            return self._to_proto(cached)

        # 再查数据库
        async with self._session_factory() as session:
            lawyer = await session.get(Lawyer, request.lawyer_id)
            if not lawyer:
                context.set_code(grpc.StatusCode.NOT_FOUND)
                context.set_details(f"Lawyer {request.lawyer_id} not found")
                return legal_pb2.LawyerResponse()
            return self._to_proto(lawyer)

    async def BatchGetLawyers(self, request, context):
        """批量获取 — 其他服务展示律师列表时需要"""
        async with self._session_factory() as session:
            stmt = select(Lawyer).where(Lawyer.id.in_(request.lawyer_ids))
            result = await session.execute(stmt)
            lawyers = result.scalars().all()
            return legal_pb2.BatchGetLawyersResponse(
                lawyers=[self._to_proto(l) for l in lawyers]
            )

    async def CheckAvailability(self, request, context):
        """可用性检查 — 预约服务需要实时查询"""
        async with self._session_factory() as session:
            stmt = (
                select(LawyerSchedule)
                .where(LawyerSchedule.lawyer_id == request.lawyer_id)
                .where(LawyerSchedule.date == request.date)
                .where(LawyerSchedule.is_booked == False)
            )
            slots = (await session.execute(stmt)).scalars().all()
            return legal_pb2.AvailabilityResponse(
                available=len(slots) > 0,
                slots=[self._slot_to_proto(s) for s in slots]
            )

from contextlib import asynccontextmanager

class TransactionManager:
    def __init__(self, session_factory):
        self._session_factory = session_factory

    @asynccontextmanager
    async def transaction(self):
        async with self._session_factory() as session:
            async with session.begin():
                yield session
                # 自动 commit / rollback

    @asynccontextmanager
    async def read_only(self):
        """只读操作，不开事务，提升性能"""
        async with self._session_factory() as session:
            session.sync_session.execute(text("SET TRANSACTION READ ONLY"))
            yield session

# ✅ app/main.py 完善 lifespan

@asynccontextmanager
async def lifespan(app: FastAPI):
    # ========== 启动阶段 ==========
    logger.info("Legal service starting...")

    # 1. 数据库
    await init_db()

    # 2. Redis
    redis_pool = await init_redis()

    # 3. Kafka Producer
    kafka_producer = await init_kafka_producer()

    consumer_shutdown = asyncio.Event()
    consumer_task = asyncio.create_task(
        start_kafka_consumer(consumer_shutdown)
    )

    # 6. Outbox Publisher（定期扫描发送）
    outbox_shutdown = asyncio.Event()
    outbox_task = asyncio.create_task(
        outbox_publisher.run(outbox_shutdown)
    )

    # 7. Consul 注册
    await register_to_consul()

    logger.info("Legal service started successfully ✅")

    yield

    # ========== 关闭阶段（按依赖逆序）==========
    logger.info("Legal service shutting down...")

    # 1. Consul 注销（立即停止接收新流量）
    await deregister_from_consul()
    logger.info("  ✓ Consul deregistered")

    # 2. 停止 Kafka Consumer（不再消费新消息）
    consumer_shutdown.set()
    try:
        await asyncio.wait_for(consumer_task, timeout=10)
    except asyncio.TimeoutError:
        consumer_task.cancel()
    logger.info("  ✓ Kafka consumer stopped")

    # 3. 停止 gRPC（给 5s grace period 完成在途请求）
    await grpc_server.stop(grace=5)
    logger.info("  ✓ gRPC server stopped")

    # 4. 发送剩余 Outbox 事件
    outbox_shutdown.set()
    try:
        await asyncio.wait_for(outbox_task, timeout=15)
    except asyncio.TimeoutError:
        outbox_task.cancel()
    logger.info("  ✓ Outbox publisher stopped")

    # 5. Flush Kafka Producer
    await kafka_producer.flush()
    await kafka_producer.stop()
    logger.info("  ✓ Kafka producer stopped")

    # 6. 关闭 Redis
    await redis_pool.close()
    logger.info("  ✓ Redis closed")

    # 7. 关闭数据库（最后关，因为其他组件可能依赖）
    await engine.dispose()
    logger.info("  ✓ Database closed")

    logger.info("Legal service shutdown complete ✅")

from typing import TypeVar, Generic, List, Optional
from pydantic import BaseModel, Field
from fastapi import Query

T = TypeVar("T")

class PaginationParams:
    """注入式分页参数"""
    def __init__(
        self,
        page: int = Query(1, ge=1, le=10000, description="页码"),
        page_size: int = Query(20, ge=1, le=100, description="每页条数"),
        sort_by: Optional[str] = Query(None, description="排序字段"),
        sort_order: str = Query("desc", pattern="^(asc|desc)$"),
    ):
        self.page = page
        self.page_size = page_size
        self.offset = (page - 1) * page_size
        self.sort_by = sort_by
        self.sort_order = sort_order

class PaginatedResponse(BaseModel, Generic[T]):
    """统一分页响应格式"""
    items: List[T]
    total: int
    page: int
    page_size: int
    total_pages: int
    has_next: bool
    has_prev: bool

    @classmethod
    def create(cls, items: List[T], total: int, params: PaginationParams):
        total_pages = (total + params.page_size - 1) // params.page_size
        return cls(
            items=items,
            total=total,
            page=params.page,
            page_size=params.page_size,
            total_pages=total_pages,
            has_next=params.page < total_pages,
            has_prev=params.page > 1,
        )

### 11. 预约 + 排班 需要并发安全

```python
# ❌ 高并发下多人同时预约同一时段
# 即使有幂等性中间件，两个不同用户用不同的 Idempotency-Key
# 预约同一时段仍然可能出现超卖

# ✅ 分布式锁 + 数据库行锁双保险

class AppointmentService:
    async def create_appointment(self, data, db, redis):
        lock_key = f"appointment_lock:{data.schedule_id}"

        # 第一层：Redis 分布式锁（快速拦截）
        lock = await redis.set(lock_key, "1", nx=True, ex=10)
        if not lock:
            raise BusinessError("L30902", "该时段正在被预约，请稍后再试")

        try:
            async with db.begin():
                # 第二层：数据库行锁（最终一致性保障）
                schedule = await db.execute(
                    select(LawyerSchedule)
                    .where(LawyerSchedule.id == data.schedule_id)
                    .with_for_update()  # SELECT ... FOR UPDATE
                )
                schedule = schedule.scalar_one_or_none()

                if not schedule:
                    raise BusinessError("L40401", "排班不存在")
                if schedule.is_booked:
                    raise BusinessError("L30901", "该时段已被预约")

                # 标记已占用 + 创建预约（原子操作）
                schedule.is_booked = True
                appointment = Appointment(**data.dict())
                db.add(appointment)

                # Outbox 事件
                db.add(OutboxEvent(
                    event_type="appointment.created",
                    payload={
                        "appointment_id": str(appointment.id),
                        "lawyer_id": str(schedule.lawyer_id),
                        "user_id": str(data.user_id),
                    }
                ))
        finally:
            await redis.delete(lock_key)
```

# ❌ 咨询的聊天消息如果无限制，单个咨询可能产生上万条消息

class ConsultationService:

        # 检查消息数量
        count = await db.scalar(
            select(func.count())
            .where(ChatMessage.consultation_id == consultation_id)
        )
        if count >= self.MAX_MESSAGES_PER_CONSULTATION:
            raise BusinessError("L30301", "该咨询消息数已达上限，请创建新咨询")

        msg = ChatMessage(consultation_id=consultation_id, **data.dict())
        db.add(msg)
        await db.commit()
        return msg

# ✅ 查询消息时必须分页 + 游标分页（大数据量更高效）
@router.get("/consultations/{id}/messages")
async def list_messages(
    id: UUID,
    cursor: Optional[int] = Query(None, description="上一页最后一条消息ID"),
    limit: int = Query(50, ge=1, le=100),
    db: AsyncSession = Depends(get_db),
):
    stmt = (
        select(ChatMessage)
        .where(ChatMessage.consultation_id == id)
    )
    if cursor:
        stmt = stmt.where(ChatMessage.id > cursor)
    stmt = stmt.order_by(ChatMessage.id.asc()).limit(limit)

    messages = (await db.execute(stmt)).scalars().all()
    return {
        "messages": messages,
        "next_cursor": messages[-1].id if messages else None,
        "has_more": len(messages) == limit,
    }
