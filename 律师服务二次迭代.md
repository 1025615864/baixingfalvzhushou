### 4. 缺少数据库事务管理策略

当前每个 service 方法可能各自管理 session，容易出现 **部分提交** 问题。

```python
# ✅ 添加统一事务管理器 app/database.py 中增加
from contextlib import asynccontextmanager

class TransactionManager:
    """统一事务管理，确保原子性操作"""

    def __init__(self, session_factory):
        self._session_factory = session_factory

    @asynccontextmanager
    async def transaction(self):
        async with self._session_factory() as session:
            async with session.begin():
                try:
                    yield session
                    # begin() 上下文自动 commit
                except Exception:
                    # begin() 上下文自动 rollback
                    raise

# ✅ 使用示例：创建预约（需要同时更新排班+创建预约+发事件）
async def create_appointment(self, data: AppointmentCreate):
    async with self.tx.transaction() as session:
        # 1. 检查排班可用性
        schedule = await session.get(Schedule, data.schedule_id, with_for_update=True)
        if not schedule or schedule.is_booked:
            raise BusinessError("L40901", "该时段已被预约")

        # 2. 标记排班已占用
        schedule.is_booked = True

        # 3. 创建预约记录
        appointment = Appointment(**data.dict())
        session.add(appointment)

        # 全部成功才提交，任一失败全部回滚
```

### 5. 缺少幂等性设计

预约、创建咨询等操作没有防重复提交机制：

```python
# ✅ 方案一：请求级幂等键（推荐）
# app/middleware/idempotency.py

class IdempotencyMiddleware(BaseHTTPMiddleware):
    """基于 Idempotency-Key 请求头的幂等性保障"""

    def __init__(self, app, redis: Redis):
        super().__init__(app)
        self.redis = redis

    async def dispatch(self, request: Request, call_next):
        if request.method not in ("POST", "PUT", "PATCH"):
            return await call_next(request)

        idempotency_key = request.headers.get("Idempotency-Key")
        if not idempotency_key:
            return await call_next(request)

        cache_key = f"idempotency:{idempotency_key}"

        # 检查是否已处理过
        cached = await self.redis.get(cache_key)
        if cached:
            return JSONResponse(
                content=json.loads(cached),
                headers={"X-Idempotent-Replayed": "true"}
            )

        # 加分布式锁防并发
        lock_key = f"idempotency_lock:{idempotency_key}"
        acquired = await self.redis.set(lock_key, "1", nx=True, ex=10)
        if not acquired:
            return JSONResponse(status_code=409, content={"detail": "请求处理中"})

        try:
            response = await call_next(request)
            # 缓存结果 24 小时
            body = [chunk async for chunk in response.body_iterator]
            await self.redis.set(cache_key, b"".join(body), ex=86400)
            return Response(
                content=b"".join(body),
                status_code=response.status_code,
                headers=dict(response.headers),
            )
        finally:
            await self.redis.delete(lock_key)
```

```python
# app/models/outbox.py

class OutboxEvent(Base):
    __tablename__ = "outbox_events"

    id = Column(UUID, primary_key=True, default=uuid4)
    event_type = Column(String(100), nullable=False, index=True)
    payload = Column(JSONB, nullable=False)
    status = Column(
        String(20),
        default="pending",
        index=True,
        comment="pending|published|failed"
    )
    retry_count = Column(Integer, default=0)
    created_at = Column(DateTime, server_default=func.now())
    published_at = Column(DateTime, nullable=True)

# app/events/outbox_publisher.py
class OutboxPublisher:
    """定期扫描 outbox 表发送到 Kafka"""

    async def publish_pending_events(self):
        async with self.session_factory() as session:
            stmt = (
                select(OutboxEvent)
                .where(OutboxEvent.status == "pending")
                .where(OutboxEvent.retry_count < 5)
                .order_by(OutboxEvent.created_at)
                .limit(100)
                .with_for_update(skip_locked=True)  # 多实例安全
            )
            events = (await session.execute(stmt)).scalars().all()

            for event in events:
                try:
                    await self.kafka_producer.send(
                        topic=f"legal.{event.event_type}",
                        value=event.payload
                    )
                    event.status = "published"
                    event.published_at = datetime.utcnow()
                except Exception as e:
                    event.retry_count += 1
                    if event.retry_count >= 5:
                        event.status = "failed"
                    logger.error(f"Outbox publish failed: {e}")

            await session.commit()
```

```python
async def get_rate_limit(path: str, user_id: str, redis: Redis) -> bool:
    for prefix, (limit, window) in RATE_LIMIT_RULES.items():
        if path.startswith(prefix):
            key = f"rate:{prefix}:{user_id}"
            current = await redis.incr(key)
```

```python
from pydantic import BaseModel, Field
from typing import TypeVar, Generic, List
from fastapi import Query

T = TypeVar("T")

class PaginationParams:
    """统一分页参数依赖注入"""
    def __init__(
        self,
        page: int = Query(1, ge=1, description="页码"),
        page_size: int = Query(20, ge=1, le=100, description="每页条数"),
        sort_by: str = Query("created_at", description="排序字段"),
        sort_order: str = Query("desc", regex="^(asc|desc)$"),
    ):
        self.page = page
        self.page_size = page_size
        self.offset = (page - 1) * page_size
        self.sort_by = sort_by
        self.sort_order = sort_order

class PaginatedResponse(BaseModel, Generic[T]):
    """统一分页响应"""
    items: List[T]
    total: int
    page: int
    page_size: int
    total_pages: int

    @classmethod
    def create(cls, items: List[T], total: int, params: PaginationParams):
        return cls(
            items=items,
            total=total,
            page=params.page,
            page_size=params.page_size,
            total_pages=(total + params.page_size - 1) // params.page_size,
        )

# ✅ 路由中使用
```
