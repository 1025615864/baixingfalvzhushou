### 4. JWT HS256 在微服务架构中的致命缺陷

```python
# ❌ 当前：HS256 对称加密
JWT_ALGORITHM = "HS256"
JWT_SECRET_KEY = "your-secret-key"

# 问题：
# 1. 每个需要验证 Token 的服务都必须持有 SECRET_KEY
# 2. 任何一个服务泄露密钥，所有服务的认证全部失效
# 3. 任何持有密钥的服务都能伪造 Token
```

```python
# ✅ 改进：生产环境强制使用 RS256 非对称加密

# app/core/jwt_keys.py
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import serialization

class JWTKeyManager:
    """
    RS256: user-service 持有私钥（签发+验证）
           其他服务只持有公钥（仅验证，无法伪造）
    """

    def __init__(self, settings):
        self.algorithm = settings.JWT_ALGORITHM  # 应为 "RS256"

        if self.algorithm == "RS256":
            # 私钥：仅 user-service 持有
            self._private_key = serialization.load_pem_private_key(
                settings.JWT_PRIVATE_KEY.encode(),
                password=None,
            )
            # 公钥：可分发给所有服务
            self._public_key = self._private_key.public_key()
        else:
            self._secret = settings.JWT_SECRET_KEY

    def encode(self, payload: dict) -> str:
        """签发 Token（仅 user-service 调用）"""
        if self.algorithm == "RS256":
            return jwt.encode(payload, self._private_key, algorithm="RS256")
        return jwt.encode(payload, self._secret, algorithm="HS256")

    def decode(self, token: str) -> dict:
        """验证 Token"""
        if self.algorithm == "RS256":
            return jwt.decode(token, self._public_key, algorithms=["RS256"])
        return jwt.decode(token, self._secret, algorithms=["HS256"])

    def get_public_key_pem(self) -> str:
        """导出公钥，供其他服务使用"""
        return self._public_key.public_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PublicFormat.SubjectPublicKeyInfo,
        ).decode()

# ✅ 提供公钥分发接口
@router.get("/.well-known/jwks.json")
async def get_jwks():
    """其他服务可以自动获取公钥验证 Token"""
    return {"keys": [jwt_key_manager.get_jwk()]}

# ✅ 其他微服务(legal-service 等)的 auth 中间件
# 只需要公钥，不需要私钥
class TokenVerifier:
    def __init__(self):
        self._public_key = self._fetch_public_key()
```

### 5. Refresh Token 缺少轮换机制 — Token 被盗无法感知

```python
# ❌ 当前：refresh_token 7天内可反复使用，被盗后攻击者可持续刷新

# ✅ 改进：Refresh Token Rotation（每次刷新生成新的 refresh_token）
class TokenService:

    async def refresh_tokens(self, refresh_token: str, redis: Redis) -> dict:
        # 1. 验证 refresh_token
        payload = self.decode(refresh_token)
        user_id = payload["sub"]
        token_family = payload["family"]  # Token 家族 ID
        jti = payload["jti"]              # 当前 Token ID

        # 2. 检查此 token 是否已使用过（重放检测）
        used_key = f"refresh_used:{jti}"
        if await redis.exists(used_key):
            # ⚠️ 重放攻击！吊销整个 Token 家族
            await self._revoke_token_family(redis, token_family)
            await self.audit_service.log(
                user_id=user_id,
                action="TOKEN_REPLAY_DETECTED",
                status="critical"
            )
            raise SecurityError("U04003", "检测到 Token 重放攻击，已吊销所有会话")

        # 3. 标记当前 token 为已使用
        await redis.set(used_key, "1", ex=7 * 86400)

        # 4. 生成新的 Token 对
        new_jti = str(uuid4())
        new_access = self.create_access_token(user_id)
        new_refresh = self.create_refresh_token(
            user_id,
            family=token_family,  # 保持同一家族
            jti=new_jti
        )

        return {
            "access_token": new_access,
            "refresh_token": new_refresh,  # ✅ 新的 refresh_token
            "token_type": "bearer"
        }

    async def _revoke_token_family(self, redis: Redis, family: str):
        """吊销整个 Token 家族"""
        await redis.set(f"family_revoked:{family}", "1", ex=7 * 86400)
```

```python
# ✅ 方案一：利用 FastAPI lifespan + asyncio 轻量调度

# app/tasks/scheduler.py
import asyncio
from datetime import datetime

class TaskScheduler:
    def __init__(self):
        self._tasks: list[asyncio.Task] = []
        self._running = True

    async def start(self):
        self._tasks.append(
            asyncio.create_task(self._run_periodic(
                name="account_deletion",
                func=self._process_expired_deletions,
                interval_seconds=3600,  # 每小时检查
            ))
        )
        self._tasks.append(
            asyncio.create_task(self._run_periodic(
                name="membership_expiry",
                func=self._check_membership_expiry,
                interval_seconds=3600,
            ))
        )

    async def stop(self):
        self._running = False
        for task in self._tasks:
            task.cancel()
        await asyncio.gather(*self._tasks, return_exceptions=True)

    async def _run_periodic(self, name: str, func, interval_seconds: int):
        while self._running:
            try:
                logger.info(f"Running scheduled task: {name}")
                await func()
            except Exception as e:
                logger.error(f"Task {name} failed: {e}")
            await asyncio.sleep(interval_seconds)

    async def _process_expired_deletions(self):
        """处理超过 7 天冷静期的注销请求"""
        async with self.session_factory() as session:
            cutoff = datetime.utcnow() - timedelta(days=7)
            stmt = (
                select(User)
                .where(User.deletion_requested_at.isnot(None))
                .where(User.deletion_requested_at <= cutoff)
                .where(User.status != "deleted")
            )
            users = (await session.execute(stmt)).scalars().all()
            for user in users:
                await self.deletion_service.execute_deletion(session, user)
                logger.info(f"Account deleted after cooling period: user_id={user.id}")

# app/main.py
@asynccontextmanager
async def lifespan(app: FastAPI):
    scheduler = TaskScheduler()
    await scheduler.start()
    yield
    await scheduler.stop()
```

### 8. 密码重置流程安全加固

```python
# ✅ 完整的安全密码重置流程

class PasswordResetService:
    RESET_TOKEN_EXPIRE = 15 * 60  # 15 分钟过期
    MAX_ATTEMPTS = 3              # 同一 token 最多尝试 3 次

    async def request_reset(self, phone_or_email: str, redis: Redis, db: AsyncSession):
        user = await self._find_user(db, phone_or_email)

        # 防枚举：无论用户是否存在，都返回相同响应
        if not user:
            logger.info(f"Password reset for non-existent user: {phone_or_email}")
            return {"message": "如果该账号存在，重置链接已发送"}

        # 频率限制：同一用户 5 分钟内只能请求一次
        rate_key = f"pwd_reset_rate:{user.id}"
        if await redis.exists(rate_key):
            return {"message": "如果该账号存在，重置链接已发送"}  # 不暴露频率限制
        await redis.set(rate_key, "1", ex=300)

        # 生成一次性 token
        token = secrets.token_urlsafe(32)
        await redis.set(
            f"pwd_reset:{token}",
            json.dumps({"user_id": user.id, "attempts": 0}),
            ex=self.RESET_TOKEN_EXPIRE,
        )

        # 发送（邮件/短信）
        await self._send_reset_link(user, token)
        return {"message": "如果该账号存在，重置链接已发送"}

    async def confirm_reset(
        self, token: str, new_password: str, redis: Redis, db: AsyncSession
    ):
        cache_key = f"pwd_reset:{token}"
        data = await redis.get(cache_key)
        if not data:
            raise AuthError("U01010", "重置链接无效或已过期")

        data = json.loads(data)

        # 尝试次数限制
        if data["attempts"] >= self.MAX_ATTEMPTS:
            await redis.delete(cache_key)
            raise AuthError("U01011", "尝试次数过多，请重新获取重置链接")

        data["attempts"] += 1
        await redis.set(cache_key, json.dumps(data), keepttl=True)

        # 密码强度校验
        password_policy.validate(new_password)

        # 不能与最近 5 次密码相同
        await self._check_password_history(db, data["user_id"], new_password)

        # 更新密码
        user = await db.get(User, data["user_id"])
        user.hashed_password = hash_password(new_password)
        await db.commit()

        # 销毁 token（一次性使用）
        await redis.delete(cache_key)

        # 吊销该用户所有现有 Token
        await self._revoke_all_tokens(redis, user.id)

        # 审计
        await self.audit_service.log(user.id, "PASSWORD_RESET", status="success")
```

```python
# ✅ 加入 scheduler 的定时检查
async def _check_membership_expiry(self):
    """每小时检查会员过期"""
    async with self.session_factory() as session:
        now = datetime.utcnow()

        # 查找已过期但状态未更新的 VIP 用户

            # 发送事件通知
            await self.kafka_producer.send("user.membership.expired", {
                "user_id": user.id,
                "previous_role": old_role,
                "expired_at": now.isoformat(),
            })

            logger.info(f"Membership expired: user_id={user.id}, {old_role} → user")

        if expired_users:
            await session.commit()
            logger.info(f"Processed {len(expired_users)} expired memberships")
```

```python
# ✅ 改进：Null Object 模式 + 本地事件存储 fallback

class KafkaProducerWrapper:
    def __init__(self, enabled: bool, bootstrap_servers: str):
        self._enabled = enabled
        self._producer = None
        if enabled:
            self._producer = AIOKafkaProducer(bootstrap_servers=bootstrap_servers)

    async def send(self, topic: str, value: dict):
        if self._producer:
            try:
                await self._producer.send_and_wait(
                    topic,
                    value=json.dumps(value).encode()
                )
                return
            except Exception as e:
                logger.error(f"Kafka send failed: {e}")

        # Fallback：写入数据库 outbox 表
        logger.warning(f"Kafka unavailable, saving to outbox: {topic}")
        await self._save_to_outbox(topic, value)

    async def _save_to_outbox(self, topic: str, value: dict):
        """即使 Kafka 宕机或未启用，事件也不会丢失"""
        async with self.session_factory() as session:
            outbox = OutboxEvent(
                topic=topic,
                payload=value,
                status="pending"
            )
            session.add(outbox)
            await session.commit()
```
